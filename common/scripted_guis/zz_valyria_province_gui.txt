zz_valyria_province ={
	scope = province 	
	is_valid = {
		scope:player = {
			gold >= 1500
		}
		county.holder = scope:player
	}
	is_shown = {
		always = yes
	}
	effect = {
		add_province_modifier = zz_redux_extra_district_slot # 1.21: slots are district slots
		scope:player = {
			remove_short_term_gold = {
				gold = {
					value = 1500
				}
			}
		}
	}
}

zz_valyria_swap_holding_type ={
	scope = province 	
	saved_scopes = {
		holding_type 
		player
	}
	is_valid = {
		scope:player = {
			gold >= 2500
		}
	}

	effect = {
		if = { 
			limit = { 
				scope:holding_type ?=  trait:holding_type_03 
			}
			set_holding_type = city_holding
		}
		if = { 
			limit = { 
				scope:holding_type ?= trait:holding_type_02
			}
			set_holding_type = church_holding
		}
		if = { 
			limit = { 
				scope:holding_type ?= trait:holding_type_01
			}
			set_holding_type = castle_holding
		}
		scope:player = {
			remove_short_term_gold = {
				gold = {
					value = 2500
				}
			}
		}
	}
}

zz_valyria_dragon_visit = {
	scope = province 	
	is_shown = {
		scope:actor = { 
			has_trait = zz_magister
		}
	}
	is_valid = { 
		scope:actor = { 
			is_available = yes
		}
	}
	effect = {
		scope:actor = {
			start_travel_plan = {
				destination = ROOT
				on_arrival_event = zz_valyria_province.0001
				on_arrival_destinations = all_but_last
			}
		}
		
	}
}

zz_pw_province_cast_click = {
	scope = character
	is_shown = {
		zz_valyria_is_mage = yes 
	}
	is_valid = {

	}
	effect = {
		save_scope_as = actor 
		set_variable = {
			name = zz_selected_mystery_province 
			value = scope:selected_province
		}
		set_variable = {
			name = zz_selected_mystery_county 
			value = scope:selected_province.county
		}
		# Provinces with no holding built have no owner of their own, so use the county's holder
		scope:selected_province = {
			if = {
				limit = {
					exists = province_owner
				}
				province_owner = {
					save_scope_as = recipient
				}
			}
			else = {
				county.holder ?= {
					save_scope_as = recipient
				}
			}
		}
		if = {
			limit = {
				exists = scope:recipient
			}
			zz_ui_setup_effect = yes
		}
	}
}
